package main

import (
	"context"
	"errors"
	"io"
	"net/http"
	"os"
	"strings"
	"testing"
	"time"
)

type reviewTransport func(*http.Request) (*http.Response, error)

func (f reviewTransport) RoundTrip(r *http.Request) (*http.Response, error) { return f(r) }

func reviewRun(t *testing.T, args []string, envTimeout string, transport http.RoundTripper) (int, string, string) {
	t.Helper()
	t.Setenv("CLA_TOKEN", "")
	t.Setenv("CLA_TIMEOUT", envTimeout)
	oldOut, oldErr, oldTransport := os.Stdout, os.Stderr, http.DefaultTransport
	out, err := os.CreateTemp(t.TempDir(), "out")
	if err != nil {
		t.Fatal(err)
	}
	defer out.Close()
	errOut, err := os.CreateTemp(t.TempDir(), "err")
	if err != nil {
		t.Fatal(err)
	}
	defer errOut.Close()
	defer func() { os.Stdout, os.Stderr, http.DefaultTransport = oldOut, oldErr, oldTransport }()
	os.Stdout, os.Stderr, http.DefaultTransport = out, errOut, transport
	code := run(args)
	os.Stdout, os.Stderr = oldOut, oldErr
	out.Seek(0, 0)
	errOut.Seek(0, 0)
	stdout, _ := io.ReadAll(out)
	stderr, _ := io.ReadAll(errOut)
	return code, string(stdout), string(stderr)
}

func TestReviewHelpExitsSuccessfully(t *testing.T) {
	code, stdout, stderr := reviewRun(t, []string{"-h"}, "", reviewTransport(func(r *http.Request) (*http.Response, error) {
		t.Error("help reached network")
		return nil, errors.New("unexpected network")
	}))
	t.Logf("code=%d stdout=%q stderr=%q", code, stdout, stderr)
	if code != 0 {
		t.Errorf("help must succeed, got exit %d", code)
	}
}

func TestReviewMalformedTimeoutEnvironmentHasDiagnostic(t *testing.T) {
	code, stdout, stderr := reviewRun(t, []string{"owner", "repo"}, "not-a-duration", reviewTransport(func(r *http.Request) (*http.Response, error) {
		t.Error("invalid timeout reached network")
		return nil, errors.New("unexpected network")
	}))
	t.Logf("code=%d stdout=%q stderr=%q", code, stdout, stderr)
	if code == 0 || !strings.Contains(stderr, "CLA_TIMEOUT") {
		t.Errorf("want nonzero exit and CLA_TIMEOUT diagnostic")
	}
}

func TestReviewStalledRequestHonorsConfiguredDeadline(t *testing.T) {
	start := time.Now()
	called := false
	code, stdout, stderr := reviewRun(t, []string{"owner", "repo"}, "10ms", reviewTransport(func(r *http.Request) (*http.Response, error) {
		called = true
		select {
		case <-r.Context().Done():
			if !errors.Is(r.Context().Err(), context.DeadlineExceeded) {
				t.Error("unexpected cancellation cause")
			}
			return nil, r.Context().Err()
		case <-time.After(time.Second):
			t.Error("request not canceled within one second")
			return nil, errors.New("test safety timeout")
		}
	}))
	elapsed := time.Since(start)
	t.Logf("code=%d elapsed=%s stdout=%q stderr=%q", code, elapsed, stdout, stderr)
	if !called || code == 0 || !strings.Contains(stderr, "deadline exceeded") || elapsed > time.Second {
		t.Error("deadline contract not satisfied")
	}
}
func TestReviewVerdictControls(t *testing.T) {
	for _, tc := range []struct {
		name              string
		positive, partial bool
		wantCode          int
	}{
		{"clean_negative", false, false, 0},
		{"clean_positive", true, false, 0},
		{"partial_unknown", false, true, 3},
		{"partial_positive", true, true, 3},
	} {
		t.Run(tc.name, func(t *testing.T) {
			code, stdout, stderr := reviewRun(t, []string{"audit-owner", "audit-repo"}, "", reviewTransport(func(r *http.Request) (*http.Response, error) {
				status, body := http.StatusOK, ""
				switch r.URL.Path {
				case "/rate_limit":
					body = `{"resources":{"core":{"remaining":1000,"limit":5000}}}`
				case "/repos/audit-owner/audit-repo":
					body = `{"default_branch":"main"}`
				case "/repos/audit-owner/audit-repo/git/trees/main":
					body = `{"tree":[],"truncated":false}`
					if tc.positive {
						body = `{"tree":[{"path":"README.md","type":"blob","sha":"readme","mode":"100644"}],"truncated":false}`
					}
				case "/repos/audit-owner/audit-repo/git/blobs/readme":
					body = `{"encoding":"base64","content":"Q0xB"}`
				case "/repos/audit-owner/audit-repo/pulls":
					body = `[]`
					if tc.partial {
						status, body = http.StatusInternalServerError, `{"message":"fixture PR failure"}`
					}
				default:
					t.Errorf("unexpected request path: %s", r.URL.Path)
					return nil, errors.New("unexpected fixture request")
				}
				return &http.Response{StatusCode: status, Header: http.Header{"Content-Type": {"application/json"}}, Body: io.NopCloser(strings.NewReader(body)), Request: r}, nil
			}))
			t.Logf("code=%d stdout=%q stderr=%q", code, stdout, stderr)
			if code != tc.wantCode {
				t.Errorf("want code %d got %d", tc.wantCode, code)
			}
			if tc.partial {
				if !strings.Contains(stdout, "INCOMPLETE") || !strings.Contains(stdout, "PRs could not be checked") || !strings.Contains(stderr, "fixture PR failure") {
					t.Error("partial evidence or diagnostics missing")
				}
				if strings.Contains(stdout, "DOES NOT need a CLA") {
					t.Error("incomplete scan rendered definitive negative")
				}
			} else {
				if stderr != "" {
					t.Error("clean scan printed diagnostics")
				}
				if strings.Contains(stdout, "INCOMPLETE") {
					t.Error("clean scan marked incomplete")
				}
				if !tc.positive && !strings.Contains(stdout, "DOES NOT need a CLA") {
					t.Error("missing clean negative verdict")
				}
			}
			if tc.positive && !strings.Contains(stdout, "DOES need a CLA") {
				t.Error("positive evidence was lost")
			}
		})
	}
}
